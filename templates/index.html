(async () => {
    let allData = [];
    let isFirstPage = true;

    // Configuration: 2000ms (2 seconds) wait for page load
    const LOAD_DELAY = 2000; 

    const scrapeCurrentPage = () => {
        const table = document.querySelector('table');
        if (!table) return;

        const rows = Array.from(table.querySelectorAll('tr'));
        
        rows.forEach((row) => {
            // Skip header rows on subsequent pages to avoid duplicates
            if (!isFirstPage && row.querySelector('th')) return;

            const cells = row.querySelectorAll('th, td');
            const rowData = Array.from(cells).map(cell => {
                // 1. Get text
                // 2. Replace everything that is NOT a letter (a-z) or number (0-9)
                // /[^a-zA-Z0-9]/g -> ^ means "NOT", g means "global"
                let cleanText = cell.innerText.replace(/[^a-zA-Z0-9]/g, '');
                
                return `"${cleanText}"`;
            });
            allData.push(rowData.join(","));
        });
        isFirstPage = false;
    };

    const getNextButton = () => {
        // Targets the Next button specifically
        return document.querySelector('button[cfctooltip*="Next page"]');
    };

    const downloadCSV = (content) => {
        const csvText = content.join("\r\n");
        const blob = new Blob([csvText], { type: 'text/csv;charset=utf-8;' });
        const url = URL.createObjectURL(blob);
        const link = document.createElement("a");
        link.setAttribute("href", url);
        link.setAttribute("download", "alphanumeric_table_data.csv");
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
        console.log("Process complete. Your file has been downloaded.");
    };

    // Main Scraping Loop
    while (true) {
        console.log("Extracting page data...");
        scrapeCurrentPage();

        const nextBtn = getNextButton();

        // Stop if button is missing, disabled, or aria-disabled
        if (!nextBtn || nextBtn.disabled || nextBtn.getAttribute('aria-disabled') === 'true' || nextBtn.classList.contains('cm-disabled')) {
            console.log("Reached final page.");
            break;
        }

        console.log("Clicking Next...");
        nextBtn.click();

        // Wait for table to refresh
        await new Promise(resolve => setTimeout(resolve, LOAD_DELAY));
    }

    downloadCSV(allData);
})();
